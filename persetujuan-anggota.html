async function updateStatus(item, newStatus) { if (busy) return; const action = newStatus === "approved" ? "menyetujui" : "menolak"; if (!confirm( "Yakin ingin " + action + " pendaftaran " + item.nama + "?" )) { return; } busy = true; showMessage("Sedang memproses pendaftaran..."); try { // Pastikan akun masih memiliki sesi admin. const { data: sessionData, error: sessionError } = await db.auth.getSession(); if ( sessionError || !sessionData.session || sessionData.session.user.app_metadata?.role !== "admin" ) { throw new Error("Sesi admin tidak valid. Silakan login kembali."); } // Periksa apakah pendaftaran masih menunggu. const { data: registration, error: readError } = await db .from("member_registrations") .select("id, nama, nomor_whatsapp, dusun, status, created_at") .eq("id", item.id) .eq("status", "pending") .maybeSingle(); if (readError) throw readError; if (!registration) { showMessage( "Pendaftaran sudah diproses atau tidak ditemukan." ); await loadRegistrations(); return; } // Jika disetujui, masukkan ke tabel members terlebih dahulu. if (newStatus === "approved") { const { data: existingMember, error: findError } = await db .from("members") .select("id") .eq("nomor_whatsapp", registration.nomor_whatsapp) .limit(1); if (findError) { throw new Error( "Gagal memeriksa anggota: " + findError.message ); } // Jangan masukkan nomor WhatsApp yang sudah terdaftar. if (!existingMember || existingMember.length === 0) { const tanggalDaftar = registration.created_at ? registration.created_at.slice(0, 10) : new Date().toISOString().slice(0, 10); const { error: insertError } = await db .from("members") .insert({ nama_anggota: registration.nama, nomor_whatsapp: registration.nomor_whatsapp, dusun: registration.dusun, tanggal_pendaftaran: tanggalDaftar }); if (insertError) { throw new Error( "Gagal menambahkan anggota: " + insertError.message ); } } } // Perbarui status pendaftaran setelah proses di atas berhasil. const { data: updated, error: updateError } = await db .from("member_registrations") .update({ status: newStatus }) .eq("id", registration.id) .eq("status", "pending") .select("id"); if (updateError) { throw new Error( "Data anggota mungkin sudah tersimpan, tetapi status pendaftaran gagal diperbarui: " + updateError.message ); } if (!updated || updated.length === 0) { throw new Error( "Status tidak berubah. Periksa kembali data pendaftaran." ); } showMessage( newStatus === "approved" ? "Pendaftaran " + registration.nama + " disetujui dan diproses ke tabel anggota." : "Pendaftaran " + registration.nama + " ditolak." ); await loadRegistrations(); } catch (error) { showMessage("Gagal memproses pendaftaran: " + error.message); } finally { busy = false; } }
